\newcommand{\thesisgraphic}[2][0.9\textwidth]{%
  \IfFileExists{#2}{%
    \includegraphics[width=#1]{#2}%
  }{%
    \fbox{\parbox[c][4cm][c]{#1}{\centering
      Figura pendiente de renderizado\\[0.4em]
      \texttt{\detokenize{#2}}}}%
  }%
}
